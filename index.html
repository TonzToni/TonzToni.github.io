Code Sample — Chained Hash Table Lookup (C)
Node *lookup(HashTable *ht, const char *key) {
    unsigned int idx = hash_fn(key) % ht->capacity;
    for (Node *curr = ht->buckets[idx]; curr != NULL; curr = curr->next) {
        if (strcmp(curr->key, key) == 0) return curr;
    }
    return NULL;
}
