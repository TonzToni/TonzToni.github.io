Sorting algorithm — Save Score (C#)
using System;
using System.Collections.Generic;
using System.IO;

namespace HighScoreLibrary
{
    public class HighScoreManager
    {
        string highScoreFile;
        int counter = 0;


        public HighScoreManager(String highScoreFilePath)
        {
           highScoreFile = highScoreFilePath;
        }

        public void SaveHighScore(String playerName, int moves)
        {
            // write new player score into highscore file, creates file if one doesnt exist
            using (StreamWriter writer = new StreamWriter(highScoreFile, true))
            {
                writer.WriteLine($"{playerName},{moves}");
            }
        }

        public void LoadHighScores()
        {
            // looks for a highscore file to sort, if not print as such
            if (File.Exists(highScoreFile))
            {
                DisplayHighScores();
            }
            else
            {
                Console.WriteLine("No high scores yet!");
            }
        }
        static void MergeSort(List<(string name, int score)> list)
        {
            // get length of input list and if 1 or less no sorting is done
            int length = list.Count;
            if (length <= 1) return;

            // calculate middle num of list
            int middle = length / 2;

            // create 2 lists each being half of the overall list being altered
            List<(string name, int score)> leftList = list.GetRange(0, middle);
            List<(string name, int score)> rightList = list.GetRange(middle, length - middle);

            // repeat above code for both sides until list is count 1 or below
            MergeSort(leftList);
            MergeSort(rightList);

            Merge(leftList, rightList, list);
        }

        static void Merge(List<(string name, int score)> leftList, List<(string name, int score)> rightList, List<(string name, int score)> list)
        {
            // declare size of lists
            int leftSize = leftList.Count;
            int rightSize = rightList.Count;

            // for looping
            int i = 0, l = 0, r = 0;

            // while both left and right are less than current increment add to halfed list
            while (l < leftSize && r < rightSize)
            {
                // if left[increment] is less than right[increment] left has the lowest number
                if (leftList[l].score < rightList[r].score)
                {
                    // add left to list and increment i,l
                    list[i] = leftList[l];
                    i++;
                    l++;
                }
                // else means right is larger or equal
                else
                {
                    // add right to list and increment i,r
                    list[i] = rightList[r];
                    i++;
                    r++;
                }
            }

            // cleanup process incase left and right were uneven
            while (l < leftSize)
            {
                // if any leftover in the left add to main list and increment i,l
                list[i] = leftList[l];
                i++;
                l++;
            }
            while (r < rightSize)
            {
                // if any leftover in the right add to main list and increment i,r
                list[i] = rightList[r];
                i++;
                r++;
            }
        }

        public void DisplayHighScores()
        {
            // find highscore file
            if (File.Exists(highScoreFile))
            {
                using (StreamReader reader = new StreamReader(highScoreFile))
                {
                    // create list to input name and score
                    List<(string name, int score)> list = new List<(string name, int score)>();

                    // read next line until none left to read
                    string line;
                    while ((line = reader.ReadLine()) != null)
                    {
                        // find split in line declaring what part is name and what part is score
                        string[] parts = line.Split(',');
                        if (parts.Length == 2)
                        {
                            // add name and score to list in that order
                            list.Add((parts[0], Int32.Parse(parts[1])));

                            // increment counter
                            counter++;
                        }
                    }


                    /*
                    cuts list down into individual elements through recursion,
                    then builds it back up while comparing sizes of each element to order them properly
                    */
                    // start initial merge sorting
                    MergeSort(list);

                    Console.Clear();
                    Console.WriteLine("High Scores:");

                    int scoreOutputCounter = 0;
                    // loop through and print out sorted list
                    for (int i = list.Count-1; i > 0; i--)
                    {
                        // stopper after top 10 printed out
                        if (scoreOutputCounter > 10)
                            return;

                        // print score and increment count for stopper
                        scoreOutputCounter++;
                        Console.WriteLine(list[i].name + " Moves: " + list[i].score);
                    }
                }
            }
        }


        public void EndOfGame(int score)
        {
            Console.Clear();
            Console.WriteLine("Game Over!");

            // Ask for the player's name
            Console.Write("Enter your name: ");
            string playerName = Console.ReadLine();

            // Save the high score
            SaveHighScore(playerName, score);

            // Load and display high scores
            LoadHighScores();
        }
    }
}
Portal System — Plane Render (C#)
using System;
using UnityEngine;

public class PortalRender : MonoBehaviour
{
    [Header("Portal One")]
    public GameObject portalOne;
    public GameObject portalOneRender;
    public Camera portalOneCam;

    [Header("Portal Two")]
    public GameObject portalTwo;
    public GameObject portalTwoRender;
    public Camera portalTwoCam;


    private Camera mainCamera;

    private RenderTexture entryPortalTex;
    private RenderTexture exitPortalTex;

    void Awake()
    {
        // find cameras
        mainCamera = GameObject.FindGameObjectWithTag("PlayerCamera").GetComponent();

        // create render texture for both portals, the RenderTextureFormat specifically needs to be set to match main camera
        entryPortalTex = new RenderTexture(Screen.width, Screen.height, 24, RenderTextureFormat.ARGBFloat);
        exitPortalTex = new RenderTexture(Screen.width, Screen.height, 24, RenderTextureFormat.ARGBFloat);
    }

    void Start()
    {
        // assign render texture to camera
        portalOneCam.targetTexture = exitPortalTex;
        portalTwoCam.targetTexture = entryPortalTex;

        // set portals mat texture to render texture
        portalOne.GetComponentInChildren().material.SetTexture("_MainTex", entryPortalTex);
        portalTwo.GetComponentInChildren().material.SetTexture("_MainTex", exitPortalTex);
    }

    void LateUpdate()
    {
        // sets up texture and camera befoe main player camera
        SetupPortalRender();
    }

    // sets up portal right before main camera render
    private void SetupPortalRender()
    {
            // match camera render texture resolution with game resolution
            // configure portal cam transforms
            // put zNear at the portal plane so it doesnt render objects behind portal

            // entry portal
            ResizeRenderTex(entryPortalTex, portalTwoCam);
            CameraTransform(portalTwoCam, portalOne);
            SetPortalClippingPlane(portalTwoCam, portalTwoRender.transform);
            
            // exit portal
            ResizeRenderTex(exitPortalTex, portalOneCam);
            CameraTransform(portalOneCam, portalTwo);
            SetPortalClippingPlane(portalOneCam, portalOneRender.transform);
    }

    void ResizeRenderTex(RenderTexture tex, Camera cam)
    {
        if (tex.width != Screen.width || tex.height != Screen.height)
        {
            // allows for window size changes dynamically without affecting portal
            tex.Release();
            tex.width = Screen.width;
            tex.height = Screen.height;
            cam.ResetAspect();
        }

        // keep fov the same on all cams
        cam.fieldOfView = mainCamera.fieldOfView;
    }

    void CameraTransform(Camera cam, GameObject portal)
    {
        // calculate player position as if it were child to portal
        Vector3 camPosOffset = portal.transform.InverseTransformPoint(mainCamera.transform.position);

        // set calculated position with inverted x / z so its behind portal
        cam.transform.localPosition = new Vector3(-camPosOffset.x, camPosOffset.y, -camPosOffset.z);

        // get both portal camera to copy main camera rotation reletive to there respective portals
        cam.transform.localRotation = Quaternion.Inverse(portal.transform.rotation) * mainCamera.transform.rotation;

        // apply a 180 degree offset so it faces portal
        cam.transform.localEulerAngles -= new Vector3(0, 180f, 0);
    }

    

    /**************************************************************************************
    finds and sets the cameras zNear using the Oblique projection matrix

    though id class this as a required standard algorithm, the struture came from 
    'Daniel Ilett (2019), https://danielilett.com/2019-12-18-tut4-3-matrix-matching/'
    **************************************************************************************/
    void SetPortalClippingPlane(Camera portalCam, Transform portalSurface)
    {
        // finds exact location of renderplane, the 2f hardcoded represents diference of space between portals
        Vector3 portalPlaneOffset = portalSurface.position - portalSurface.forward * (Math.Abs(portalSurface.localPosition.z) * 2);

        // create a clip plane where the render plane is and converts to world space coordinates for projection matrix
        Plane p = new Plane(-portalSurface.forward, portalPlaneOffset);
        Vector4 clipPlane = new Vector4(p.normal.x, p.normal.y, p.normal.z, p.distance);

        // creates a new projection matrix with the new zNear calculation
        Vector4 clipPlaneCameraSpace = Matrix4x4.Transpose(Matrix4x4.Inverse(portalCam.worldToCameraMatrix)) * clipPlane;

        // overrite the standard projection matrix with the new sqewed zNear at render plane
        portalCam.projectionMatrix = portalCam.CalculateObliqueMatrix(clipPlaneCameraSpace);
    }
}
Entity Framework — Interaction (C#)
using System;
using UnityEngine;

public abstract class BaseEntity : MonoBehaviour, IDamageable
{
    public EntityStats stats;

    private float internalHealth;
    public float currentHealth
    {
        get => internalHealth;
        set
        {
            // hold health value above negative
            internalHealth = Mathf.Clamp(value, 0, stats.maxHealth);

            // call Die() if health is determined to be 0 and entity is still alive
            if (internalHealth <= 0)
                Die();
        }
    }

    protected virtual void OnEnable()
    {
        // get health value from stats SO
        if (stats != null) currentHealth = stats.maxHealth;
    }

    public virtual void TakeDamage(float damage, GameObject attacker)
    {
        currentHealth -= damage;
    }

    protected virtual void Die()
    {
        PlayerHealth.instance.Rumble();

        // destroy entity from scene
        Destroy(this.gameObject);
    }
}
