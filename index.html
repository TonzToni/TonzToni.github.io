Node *lookup(HashTable *ht, const char *key) {
    unsigned int idx = hash_fn(key) % ht->capacity;
    for (Node *curr = ht->buckets[idx]; curr != NULL; curr = curr->next) {
        if (strcmp(curr->key, key) == 0) return curr;
    }
    return NULL;
}

def expectimax(board, depth, is_player_turn):
    if depth == 0 or is_terminal(board):
        return evaluate_board(board)
    if is_player_turn:
        return max(expectimax(move(board, d), depth - 1, False) for d in VALID_MOVES)
    empty = get_empty_cells(board)
    return sum(0.9 * expectimax(place(board, c, 2), depth - 1, True)
               + 0.1 * expectimax(place(board, c, 4), depth - 1, True)
               for c in empty) / len(empty)
